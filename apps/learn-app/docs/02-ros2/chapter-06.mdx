## Introduction

While topics provide continuous data streams, many robotics tasks require request-response patterns (services) or long-running operations with feedback (actions). This chapter covers ROS 2's synchronous and asynchronous communication mechanisms.

## Core Concepts

### 1. Communication Pattern Comparison

| Pattern | Use Case | Feedback | Example |
|---------|----------|----------|---------|
| **Topic** | Continuous data streams | No | Sensor readings, odometry |
| **Service** | Quick request-response | No | Get robot state, set parameter |
| **Action** | Long-running tasks | Yes | Navigation, grasping, trajectory execution |

### 2. ROS 2 Services

Services provide synchronous request-response communication:

```python
# Service Server
from example_interfaces.srv import AddTwoInts
import rclpy
from rclpy.node import Node

class MinimalService(Node):
    def __init__(self):
        super().__init__('minimal_service')
        self.srv = self.create_service(
            AddTwoInts,
            'add_two_ints',
            self.add_two_ints_callback
        )

    def add_two_ints_callback(self, request, response):
        response.sum = request.a + request.b
        self.get_logger().info(f'Request: {request.a} + {request.b} = {response.sum}')
        return response

def main():
    rclpy.init()
    service = MinimalService()
    rclpy.spin(service)
    rclpy.shutdown()
```

### 3. ROS 2 Actions

Actions enable long-running tasks with progress feedback:

```python
# Action Server
import rclpy
from rclpy.action import ActionServer
from rclpy.node import Node
from example_interfaces.action import Fibonacci

class FibonacciActionServer(Node):
    def __init__(self):
        super().__init__('fibonacci_action_server')
        self._action_server = ActionServer(
            self,
            Fibonacci,
            'fibonacci',
            self.execute_callback
        )

    async def execute_callback(self, goal_handle):
        self.get_logger().info('Executing goal...')

        # Initialize feedback
        feedback_msg = Fibonacci.Feedback()
        feedback_msg.sequence = [0, 1]

        # Generate Fibonacci sequence
        for i in range(1, goal_handle.request.order):
            # Check if goal has been canceled
            if goal_handle.is_cancel_requested:
                goal_handle.canceled()
                return Fibonacci.Result()

            # Compute next value
            feedback_msg.sequence.append(
                feedback_msg.sequence[i] + feedback_msg.sequence[i-1]
            )

            # Publish feedback
            goal_handle.publish_feedback(feedback_msg)
            await asyncio.sleep(1)

        # Return result
        goal_handle.succeed()
        result = Fibonacci.Result()
        result.sequence = feedback_msg.sequence
        return result
```

## Hands-On Tutorial: Robot State Machine with Actions

```python
# robot_controller_action.py
import rclpy
from rclpy.action import ActionServer, GoalResponse, CancelResponse
from rclpy.node import Node
from geometry_msgs.msg import Twist
from nav2_msgs.action import NavigateToPose
import asyncio

class RobotController(Node):
    def __init__(self):
        super().__init__('robot_controller')

        # Create action server
        self._action_server = ActionServer(
            self,
            NavigateToPose,
            'navigate_to_pose',
            execute_callback=self.execute_callback,
            goal_callback=self.goal_callback,
            cancel_callback=self.cancel_callback
        )

        # Publisher for velocity commands
        self.cmd_vel_pub = self.create_publisher(Twist, 'cmd_vel', 10)

        self.get_logger().info('Robot Controller Action Server Started')

    def goal_callback(self, goal_request):
        """Accept or reject goal requests"""
        self.get_logger().info('Received goal request')
        return GoalResponse.ACCEPT

    def cancel_callback(self, goal_handle):
        """Handle cancellation requests"""
        self.get_logger().info('Received cancel request')
        return CancelResponse.ACCEPT

    async def execute_callback(self, goal_handle):
        """Execute navigation to pose"""
        self.get_logger().info('Executing navigation goal...')

        feedback_msg = NavigateToPose.Feedback()
        target_pose = goal_handle.request.pose

        # Simulate navigation with progress feedback
        for progress in range(0, 101, 10):
            if goal_handle.is_cancel_requested:
                goal_handle.canceled()
                self.get_logger().info('Goal canceled')
                # Stop robot
                self.publish_stop_command()
                return NavigateToPose.Result()

            # Update feedback
            feedback_msg.distance_remaining = (100 - progress) / 10.0
            goal_handle.publish_feedback(feedback_msg)

            self.get_logger().info(f'Progress: {progress}%')
            await asyncio.sleep(0.5)

        # Goal succeeded
        goal_handle.succeed()
        result = NavigateToPose.Result()
        self.get_logger().info('Goal succeeded!')
        return result

    def publish_stop_command(self):
        """Publish zero velocity to stop robot"""
        stop_msg = Twist()
        self.cmd_vel_pub.publish(stop_msg)

def main():
    rclpy.init()
    controller = RobotController()
    rclpy.spin(controller)
    rclpy.shutdown()

if __name__ == '__main__':
    main()
```
