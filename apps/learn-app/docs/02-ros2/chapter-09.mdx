## Introduction

ROS 2 bags record all message data flowing through topics, enabling replay, analysis, and testing without hardware. This chapter covers using bags for development, debugging, and validation.

## Core Concepts

### 1. Recording Data

```bash
# Record all topics
ros2 bag record -a

# Record specific topics
ros2 bag record /camera/image_raw /laser_scan /odom

# Record with storage options
ros2 bag record -a -o my_bag_name --storage sqlite3

# Record compressed
ros2 bag record -a --compression-mode file
```

### 2. Playback and Analysis

```bash
# Get bag info
ros2 bag info my_bag_name

# Play bag
ros2 bag play my_bag_name

# Play at half speed
ros2 bag play my_bag_name --rate 0.5

# Play looping
ros2 bag play my_bag_name --loop

# Play specific topics only
ros2 bag play my_bag_name --topics /odom /laser_scan
```

## Hands-On Tutorial: Automated Testing with Bags

```python
# test_navigation_with_bag.py
import unittest
import rclpy
from rclpy.node import Node
from nav_msgs.msg import Odometry
from geometry_msgs.msg import Twist
import subprocess
import time

class NavigationBagTest(unittest.TestCase):
    @classmethod
    def setUpClass(cls):
        rclpy.init()

    @classmethod
    def tearDownClass(cls):
        rclpy.shutdown()

    def test_navigation_converges(self):
        """Test that robot reaches goal using recorded sensor data"""

        # Start bag playback in background
        bag_process = subprocess.Popen([
            'ros2', 'bag', 'play',
            'test_data/navigation_test.bag',
            '--topics', '/laser_scan', '/odom'
        ])

        # Create test node
        node = NavigationTestNode()

        # Spin for 30 seconds
        start_time = time.time()
        while time.time() - start_time < 30:
            rclpy.spin_once(node, timeout_sec=0.1)

        # Check results
        self.assertTrue(node.reached_goal, "Robot failed to reach goal")
        self.assertLess(
            node.max_velocity,
            2.0,
            "Robot exceeded max velocity constraint"
        )

        # Cleanup
        bag_process.terminate()
        node.destroy_node()

class NavigationTestNode(Node):
    def __init__(self):
        super().__init__('navigation_test_node')
        self.reached_goal = False
        self.max_velocity = 0.0

        self.odom_sub = self.create_subscription(
            Odometry,
            '/odom',
            self.odom_callback,
            10
        )

        self.cmd_vel_sub = self.create_subscription(
            Twist,
            '/cmd_vel',
            self.cmd_vel_callback,
            10
        )

    def odom_callback(self, msg):
        # Check if near goal
        x = msg.pose.pose.position.x
        y = msg.pose.pose.position.y
        distance_to_goal = ((x - 5.0)**2 + (y - 3.0)**2)**0.5

        if distance_to_goal < 0.1:
            self.reached_goal = True

    def cmd_vel_callback(self, msg):
        velocity = (msg.linear.x**2 + msg.linear.y**2)**0.5
        self.max_velocity = max(self.max_velocity, velocity)

if __name__ == '__main__':
    unittest.main()
```
