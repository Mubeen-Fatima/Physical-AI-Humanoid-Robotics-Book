## Core Concepts

### 1. Isaac Sim vs Traditional Simulators

| Feature | Gazebo | Unity | Isaac Sim |
|---------|---------|-------|-----------|
| **Physics** | CPU (ODE) | CPU (PhysX) | **GPU (PhysX)** |
| **Rendering** | OGRE | HDRP | **RTX Ray Tracing** |
| **Scale** | ~10 robots | ~50 robots | **1000+ robots** |
| **AI Integration** | Limited | ML-Agents | **Omniverse AI** |

### 3. Basic Isaac Sim Scene

```python
# isaac_sim_scene.py
from omni.isaac.kit import SimulationApp
simulation_app = SimulationApp({"headless": False})

from omni.isaac.core import World
from omni.isaac.core.objects import DynamicCuboid
from omni.isaac.core.robots import Robot
import numpy as np

# Create world
world = World()

# Add ground plane
world.scene.add_default_ground_plane()

# Add robot
robot = Robot(
    prim_path="/World/Franka",
    name="franka",
    usd_path="/Isaac/Robots/Franka/franka.usd"
)
world.scene.add(robot)

# Add object to manipulate
cube = DynamicCuboid(
    prim_path="/World/Cube",
    name="cube",
    position=np.array([0.5, 0, 0.5]),
    size=0.05,
    color=np.array([1, 0, 0])
)
world.scene.add(cube)

# Reset world
world.reset()

# Simulation loop
for i in range(1000):
    world.step(render=True)

simulation_app.close()
```

### 4. ROS 2 Integration

```python
# isaac_ros2_bridge.py
from omni.isaac.kit import SimulationApp
simulation_app = SimulationApp({"headless": False})

from omni.isaac.core import World
from omni.isaac.ros2_bridge import ROS2Bridge
import rclpy

# Initialize ROS 2
rclpy.init()

# Create world
world = World()
world.scene.add_default_ground_plane()

# Create ROS 2 bridge
ros_bridge = ROS2Bridge()

# Add camera with ROS 2 publisher
camera_graph_path = "/World/Camera"
ros_bridge.create_camera_publisher(
    camera_prim_path=camera_graph_path,
    topic_name="/camera/image_raw",
    frame_id="camera_frame"
)

# Add odometry publisher
ros_bridge.create_odometry_publisher(
    robot_prim_path="/World/Robot",
    topic_name="/odom"
)

# Simulation loop
world.reset()
while simulation_app.is_running():
    world.step(render=True)

rclpy.shutdown()
simulation_app.close()
```

# synthetic_data_gen.py
from omni.isaac.synthetic_utils import SyntheticDataHelper
from omni.replicator.core import Randomizer

class DatasetGenerator:
    def __init__(self, world):
        self.world = world
        self.data_helper = SyntheticDataHelper()

    def randomize_scene(self):
        """Apply domain randomization"""
        # Randomize lighting
        Randomizer.randomize_light_intensity(min_val=500, max_val=2000)

        # Randomize object positions
        Randomizer.randomize_object_position(
            prim_path="/World/Cube",
            x_range=(-0.5, 0.5),
            y_range=(-0.5, 0.5),
            z_range=(0.1, 0.5)
        )

        # Randomize materials
        Randomizer.randomize_material(
            prim_path="/World/Cube"
        )

    def capture_frame(self):
        """Capture annotated frame"""
        # RGB image
        rgb = self.data_helper.get_rgb()

        # Depth
        depth = self.data_helper.get_depth()

        # Semantic segmentation
        semantic = self.data_helper.get_semantic_segmentation()

        # Bounding boxes
        bboxes = self.data_helper.get_bounding_boxes_2d()

            'rgb': rgb,
