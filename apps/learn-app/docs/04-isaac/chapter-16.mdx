### 3. Custom Perception Module

```cpp
// object_detector.hpp
#include "engine/alice/alice_codelet.hpp"
#include "messages/camera.capnp.h"

class ObjectDetector : public isaac::alice::Codelet {
 public:
  void start() override {
    tickPeriodically();
  }

  void tick() override {
    // Receive camera image
    auto image_proto = rx_image().getProto();

    // Run detection
    auto detections = detectObjects(image_proto);

    // Publish results
    auto detection_msg = tx_detections().initProto();
    detection_msg.setDetections(detections);
    tx_detections().publish();
  }

  // Input: camera image
  ISAAC_PROTO_RX(ImageProto, image);

  // Output: detections
  ISAAC_PROTO_TX(DetectionListProto, detections);

 private:
  std::vector<Detection> detectObjects(const ImageProto& image);
};

ISAAC_ALICE_REGISTER_CODELET(ObjectDetector);
```

### 4. Deploying to Jetson

```bash
# Build for Jetson AGX Xavier
bazel build //apps/my_robot:my_robot-pkg \
  --config=jetpack45 \
  --cpu=arm64-v8a

# Deploy to Jetson
./engine/build/deploy.sh \
  --packag my_robot-pkg \
  --host jetson.local \
  --user nvidia

# Run on Jetson
ssh nvidia@jetson.local
cd deploy/my_robot
./apps/my_robot/my_robot
```
